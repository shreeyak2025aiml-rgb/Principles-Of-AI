# Alpha-Beta Pruning Algorithm

def alphabeta(node, depth, alpha, beta, maximizing):

    if depth == 0 or node not in tree:
        return scores[node]

    if maximizing:
        best = -999

        for child in tree[node]:
            value = alphabeta(child, depth-1, alpha, beta, False)
            best = max(best, value)
            alpha = max(alpha, best)

            if beta <= alpha:
                break

        return best

    else:
        best = 999

        for child in tree[node]:
            value = alphabeta(child, depth-1, alpha, beta, True)
            best = min(best, value)
            beta = min(beta, best)

            if beta <= alpha:
                break

        return best


tree = {
    'A': ['B', 'C'],
    'B': ['D', 'E'],
    'C': ['F', 'G']
}

scores = {
    'D': 3,
    'E': 5,
    'F': 2,
    'G': 9
}

result = alphabeta('A', 2, -999, 999, True)

print("Best score:", result)
