grid = [
    [1, 2, 3, 4, 5],
    [2, 4, 6, 8, 4],
    [3, 5, 9, 7, 3],
    [1, 3, 8, 6, 2],
    [2, 4, 5, 7, 10]
]

n = len(grid)

def neighbors(state):
    r, c = state
    moves = [(1,0), (-1,0), (0,1), (0,-1)]

    result = []

    for dr, dc in moves:
        nr, nc = r + dr, c + dc

        if 0 <= nr < n and 0 <= nc < n:
            result.append((nr, nc))

    return result


current = (0, 0)

while True:

    print("Current:", current, "Value:", grid[current[0]][current[1]])

    next_states = neighbors(current)

    best = max(
        next_states,
        key=lambda x: grid[x[0]][x[1]]
    )

    if grid[best[0]][best[1]] <= grid[current[0]][current[1]]:
        break

    current = best

print("\nHill Climbing Result")
print("State:", current)
print("Maximum Value:", grid[current[0]][current[1]])
