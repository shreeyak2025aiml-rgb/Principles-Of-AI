import random
import math

grid = [
    [1, 2, 3, 4, 5],
    [2, 4, 6, 8, 4],
    [3, 5, 9, 7, 3],
    [1, 3, 8, 6, 2],
    [2, 4, 5, 7, 10]
]

n = len(grid)

def neighbors(state):
    r, c = state
    moves = [(1,0), (-1,0), (0,1), (0,-1)]

    result = []

    for dr, dc in moves:
        nr, nc = r + dr, c + dc

        if 0 <= nr < n and 0 <= nc < n:
            result.append((nr, nc))

    return result


current = (0, 0)
best = current
temperature = 100

while temperature > 0.1:

    next_state = random.choice(neighbors(current))

    current_value = grid[current[0]][current[1]]
    next_value = grid[next_state[0]][next_state[1]]

    difference = next_value - current_value

    if difference > 0:
        current = next_state

    else:
        probability = math.exp(difference / temperature)

        if random.random() < probability:
            current = next_state

    if grid[current[0]][current[1]] > grid[best[0]][best[1]]:
        best = current

    temperature *= 0.95

print("Simulated Annealing Result")
print("Best State:", best)
print("Best Value:", grid[best[0]][best[1]])
