# Local Search using Hill Climbing

import random

n = 4

board = [random.randint(0, n-1) for _ in range(n)]


def conflicts(board):
    count = 0

    for i in range(n):
        for j in range(i+1, n):

            if board[i] == board[j]:
                count += 1

            if abs(board[i]-board[j]) == abs(i-j):
                count += 1

    return count


while conflicts(board) != 0:

    best_board = board[:]
    best_score = conflicts(board)

    for col in range(n):
        for row in range(n):

            new_board = board[:]
            new_board[col] = row

            score = conflicts(new_board)

            if score < best_score:
                best_score = score
                best_board = new_board

    if best_score == conflicts(board):
        board = [random.randint(0, n-1) for _ in range(n)]
    else:
        board = best_board


print("Solution:", board)

for row in range(n):
    for col in range(n):
        if board[col] == row:
            print("Q", end=" ")
        else:
            print(".", end=" ")
    print()
