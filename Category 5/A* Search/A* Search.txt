import heapq

goal = "123456780"

def h(s):
    return sum(a != b and a != '0' for a, b in zip(s, goal))

def astar(start):
    pq = [(h(start), 0, start)]
    visited = set()

    while pq:
        f, g, state = heapq.heappop(pq)

        if state in visited:
            continue
        visited.add(state)

        print(state[:3], state[3:6], state[6:])

        if state == goal:
            print("Goal Reached!")
            return

        z = state.index('0')
        for d in [-3, 3, -1, 1]:
            n = z + d
            if 0 <= n < 9 and not (z % 3 == 0 and d == -1) and not (z % 3 == 2 and d == 1):
                x = list(state)
                x[z], x[n] = x[n], x[z]
                ns = ''.join(x)
                heapq.heappush(pq, (g + 1 + h(ns), g + 1, ns))

astar("123405786")
